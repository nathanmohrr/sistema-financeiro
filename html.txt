  <script>
    // IndexedDB config e funções básicas, adapte conforme seu código atual
    const DB_NAME = 'FinanceiroDB';
    const DB_VERSION = 5; // atualize versão para criar a nova store
    const STORE_OPERACOES = 'operacoes';

    let db;

    function abrirDB() {
      return new Promise((resolve, reject) => {
        const request = indexedDB.open(DB_NAME, DB_VERSION);

        request.onerror = e => reject(e.target.error);
        request.onsuccess = e => {
          db = e.target.result;
          resolve(db);
        };
        request.onupgradeneeded = e => {
          db = e.target.result;
          // Crie a store 'operacoes' se não existir
          if (!db.objectStoreNames.contains(STORE_OPERACOES)) {
            db.createObjectStore(STORE_OPERACOES, { keyPath: 'id', autoIncrement: true });
          }
          // aqui mantenha upgrade para outras stores que seu sistema já usa
        };
      });
    }

    async function salvarOperacao(operacao) {
      return new Promise((resolve, reject) => {
        const transaction = db.transaction(STORE_OPERACOES, 'readwrite');
        const store = transaction.objectStore(STORE_OPERACOES);
        const request = store.add(operacao);
        request.onsuccess = () => resolve();
        request.onerror = e => reject(e.target.error);
      });
    }

    async function carregarOperacoes() {
      return new Promise((resolve, reject) => {
        const transaction = db.transaction(STORE_OPERACOES, 'readonly');
        const store = transaction.objectStore(STORE_OPERACOES);
        const request = store.getAll();
        request.onsuccess = e => {
          const operacoes = e.target.result;
          const lista = document.getElementById('listaOperacoes');
          lista.innerHTML = '';
          operacoes.forEach(op => {
            const tr = document.createElement('tr');
            tr.innerHTML = `
              <td>${op.data}</td>
              <td>${op.codigoProduto}</td>
              <td>${op.cliente}</td>
              <td>R$ ${op.valorVenda.toFixed(2)}</td>
              <td>${op.formaPagamento}</td>
            `;
            lista.appendChild(tr);
          });
          resolve();
        };
        request.onerror = e => reject(e.target.error);
      });
    }

    document.addEventListener('DOMContentLoaded', () => {
      abrirDB().then(() => {
        carregarOperacoes();
      });

      document.getElementById('formOperacao').addEventListener('submit', async (e) => {
        e.preventDefault();

        const operacao = {
          codigoProduto: document.getElementById('codigoProduto').value,
          valorVenda: parseFloat(document.getElementById('valorVenda').value),
          formaPagamento: document.getElementById('formaPagamento').value,
          cliente: document.getElementById('cliente').value,
          data: document.getElementById('dataOperacao').value || new Date().toISOString().split('T')[0],
        };

        await salvarOperacao(operacao);
        await carregarOperacoes();
        e.target.reset();
      });
    });
  </script>
